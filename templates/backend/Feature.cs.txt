using MongoDB.Bson;
using MongoDB.Bson.Serialization.Attributes;
using MongoDB.Driver;
namespace Ecommerce.Web.Modules.Inventory;
public sealed class StockDocument {
    [BsonId, BsonRepresentation(BsonType.ObjectId)] public string Id { get; set; } = "";
    [BsonElement("stock")] public int Stock { get; set; }
}
public sealed record StockResponse(string ProductId, int Available);
public interface IInventoryReader {
    Task<StockResponse?> Get(string productId, CancellationToken cancellationToken);
}
public sealed class MongoInventoryReader(IMongoDatabase db) : IInventoryReader {
    public async Task<StockResponse?> Get(string productId, CancellationToken cancellationToken) {
        if (!ObjectId.TryParse(productId, out _)) throw new ArgumentException("Invalid product ID");
        var row = await db.GetCollection<StockDocument>("sanpham").Find(x => x.Id == productId)
            .Project(x => new StockResponse(x.Id, x.Stock)).FirstOrDefaultAsync(cancellationToken);
        return row;
    }
}
// This is a read model, not an independent stock mutation. Checkout must keep its existing transaction.
